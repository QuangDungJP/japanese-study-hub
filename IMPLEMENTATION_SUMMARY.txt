💎 COMPONENT PROPS:
─────────────────────────────────────────────────────────────────────────────

interface SidebarModuleProps {
  sections: SidebarSection[];      // Navigation sections
  logo?: React.ReactNode;           // Logo component
  header?: React.ReactNode;         // Header content (custom)
  footer?: React.ReactNode;         // Footer content
  onNavigate?: () => void;          // Navigate callback
  showDarkMode?: boolean;           // Dark mode toggle
  showLogout?: boolean;             // Logout button
  onLogout?: () => void;            // Logout handler
  userRole?: string;                // Role badge
  userName?: string;                // User display name
}

interface SidebarSection {
  label: string;
  items: SidebarItem[];
  collapsible?: boolean;
  defaultOpen?: boolean;
}

interface SidebarItem {
  name: string;
  href: string;
  icon: React.ComponentType<{ className?: string }>;
  badge?: string | number;
  description?: string;
}
